import { collection, getDocs, doc, setDoc, updateDoc, deleteDoc } from "https://www.gstatic.com/firebasejs/12.2.1/firebase-firestore.js"; // Create Event document.getElementById("createEvent").onclick = async () => { const name = document.getElementById("eventName").value.trim(); const map = document.getElementById("eventMap").value.trim(); const start = document.getElementById("startDate").value; const end = document.getElementById("endDate").value; if(!name || !map || !start || !end){ document.getElementById("eventMsg").textContent = "Fill in every field."; return; } // 前のイベントを全部終了 const oldEvents = await getDocs(collection(db,"events")); for(const e of oldEvents.docs){ await updateDoc(doc(db,"events",e.id),{ active:false }); } // 新しいイベント作成 const id = name.toLowerCase().replace(/\s+/g,"-"); await setDoc(doc(db,"events",id),{ name, map, start, end, active:true }); document.getElementById("eventMsg").textContent = "✅ Event Created!"; };
